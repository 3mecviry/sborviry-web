# Pravidla AI robotů níž jsou schválně vypsaná jménem, i když by stačil
# blok „*“. Dva důvody:
#   1) Je na první pohled vidět, že jde o vědomé rozhodnutí, ne o opomenutí.
#   2) Kdyby se v budoucnu blok „*“ zpřísnil, tyhle roboty to nezasáhne.
#
# Aktualizováno: srpen 2026. Jména robotů se mění — před další úpravou
# zkontrolujte dokumentaci dodavatelů.

User-agent: *
Allow: /

# Stránky, které do vyhledávání nepatří.
# Zápis bez koncovky pokrývá obě podoby adresy — /404 i /404.html.
Disallow: /404

# Zdrojové kostry stránek a generátor. Hotové stránky vznikají z nich,
# takže samy o sobě do vyhledávání nepatří — jinak by se stejný obsah
# nabízel dvakrát.
Disallow: /_sablony/
Disallow: /nastroje/

# Rozpracované návrhy. Složka je sice v .gitignore, ale kdyby se někdy
# omylem dostala do repozitáře, GitHub Pages ji hned zveřejní.
Disallow: /_nahledy/

# Dokumentace k webu. GitHub Pages servíruje všechno z kořene repozitáře,
# takže by se jinak dala najít i na sborviry.org — do výsledků vyhledávání
# ale nepatří, je určená správci webu, ne návštěvníkům.
# Znaky „*“ a „$“ jsou součástí standardu RFC 9309 (sekce 2.2.3),
# takže je respektuje Google, Bing, Seznam i doložení AI roboti; explicitní
# řádky pod tím jsou pojistka pro roboty, kteří zástupné znaky neumí.
Disallow: /*.md$
Disallow: /README.md
Disallow: /KONTROLA-JAZYKU.md

# POZNÁMKA KE STRÁNCE O OCHRANĚ OSOBNÍCH ÚDAJŮ:
# Dřív tu stálo „Disallow: /*ochrana-osobnich-udaju“. Zákaz v tomhle souboru
# ale indexaci nezabrání — vyhledávač smí adresu zařadit i bez obsahu, takže
# se ukáže jako holý odkaz bez popisku, což je horší než celá stránka.
# Navíc bránil robotům přečíst si značku noindex, která je přímo na stránce
# a která ji z výsledků odstraní spolehlivě — a přitom dovolí robotům
# i AI přečíst si právní údaje o sboru.
